    void finishInstallation(String result) {
        this.wasProgressDone = true;
        this.installedURL = this.actor.getContentToBeActed();
        SharedPreferences settings = getSharedPreferences(KEY_PREFS, 0);
        SharedPreferences.Editor editor = settings.edit();
        editor.putBoolean(IS_LOADED, true);
        editor.putString(INSTALL_URL, this.actor.getContentToBeActed());
        editor.commit();
        updateGUI();
    }
